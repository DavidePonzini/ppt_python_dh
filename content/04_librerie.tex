    \frametitle{Librerie}

    \begin{itemize}
        \item Insieme di funzioni già pronte scritte da altre persone
        \bigskip
        \item Esistono numerose librerie
        \item Ciascuna si occupa (generalmente) di una sola cosa
        \bigskip
        \item Per usarle nel mio codice uso il comando \boxed{\texttt{import}~\textit{libreria}}
        \begin{itemize}
            \item Una volta importate, per usare le funzionalità mi basta scrivere:\\
            \texttt{libreria.funzione}
        \end{itemize}
    \end{itemize}

\makesectionframe{Gestione librerie}

\begin{contentframe}
    \frametitle{Gestione librerie}

    \begin{itemize}
        \item Le librerie che abbiamo usato finora sono automaticamente installate in Python
        \item Possiamo anche installare nuove librerie

        \bigskip
        \item Usiamo il modulo pip di Python
        \begin{itemize}
            \item \texttt{python -m pip install nuovalibreria}
        \end{itemize}
    \end{itemize}
\end{contentframe}

\begin{contentframe}
    \frametitle{Ambienti virtuali}

    \begin{itemize}
        \item \texttt{pip} installa in maniera globale
        \item Spesso non è desiderabile
        \begin{itemize}
            \item Esempio: abbiamo due progetti che richiedono due diverse versioni della stessa libreria
        \end{itemize}

        \bigskip
        \item Soluzione: si usano gli ambienti virtuali
    \end{itemize}
\end{contentframe}

\begin{contentframe}
    \frametitle{Ambienti virtuali}

    \begin{itemize}
        \item L'ambiente virtuale è di fatto una cartella
        \item Contiene le librerie relative ad un solo progetto
        \item Si può cancellare e ricreare senza alcun problema

        \bigskip
        \item Consiglio: scrivere tutte le librerie che scaricate in un file chiamato \texttt{requirements.txt}
    \end{itemize}
\end{contentframe}

\begin{contentframe}
    \frametitle{Ambienti virtuali}

    \begin{itemize}
        \item Per creare una ambiente virtuale si usa il comando \texttt{python -m venv venv}
        \begin{itemize}
            \item Il primo \texttt{venv} è il modulo, il secondo è il nome della cartella che creeremo
        \end{itemize}

        \bigskip
        \item Per poterlo usare lo si deve prima attivare
        \begin{itemize}
            \item Windows: \texttt{./venv/Scripts/activate}
            \item Linux, Mac: \texttt{source ./venv/bin/activate}
        \end{itemize}

        \bigskip
        \item In alternativa, si può lanciare direttamente l'eseguibile all'interno dell'ambiente:
        \begin{itemize}
            \item Windows: \texttt{./venv/Scripts/python}
            \item Linux, Mac: \texttt{./venv/bin/python}
        \end{itemize}
    \end{itemize}
\end{contentframe}

    \begin{itemize}
        \item Operazioni su \textit{Dataset}

\begin{contentframe}
    \frametitle{Librerie}
    \framesubtitle{pandas - Strutture dati}

    \begin{itemize}
        \item Due strutture dati da conoscere bene

        \bigskip
        \item \textbf{DataFrame}
        \begin{itemize}
            \item Insieme di righe e colonne
            \item Rappresenta una tabella
        \end{itemize}

        \bigskip
        \item \textbf{Series}
        \begin{itemize}
            \item Sequenza di valori
            \item Generalmente rappresenta una colonna
        \end{itemize}
    \end{itemize}
\end{contentframe}

\begin{contentframe}
    \frametitle{Librerie}
    \framesubtitle{pandas -- Operazioni principali}

    \begin{itemize}
        \item Per accedere ad una colonna si usano le \texttt{[...]}
        \begin{itemize}
            \item \texttt{studenti[\textquotesingle{}nome\textquotesingle{}]}
        \end{itemize}

        \bigskip
        \item Per accedere ad una riga si usa la funzione \texttt{.iloc[...]}
        \begin{itemize}
            \item Si comporta come le \texttt{[]} nelle liste
            \item Può ritornare una riga sola
            \item Si può fare slicing, ritornando un insieme di righe
        \end{itemize}
    \end{itemize}
\end{contentframe}

\begin{contentframe}
    \frametitle{Librerie}
    \framesubtitle{pandas -- Operazioni principali}

    \begin{itemize}
        \item Creazione, modifica e cancellazione di colonne avviene in maniera simile ai dizionari
        \begin{itemize}
            \item La chiave è il nome della colonna
            \item Se non esiste, la creo
            \item Se esiste, la sovrascrivo
        \end{itemize}

        \bigskip
        \item Differenza rispetto ai dizionari
        \begin{itemize}
            \item Le azioni che indico sono automaticamente effettuate su ciascun valore
            \item \texttt{df[\textquotesingle{}col2\textquotesingle{}] = df[\textquotesingle{}col1\textquotesingle{}] + 1}
        \end{itemize}
    \end{itemize}
\end{contentframe}

\begin{contentframe}
    \frametitle{Librerie}
    \framesubtitle{pandas -- Operazioni principali}

    \begin{itemize}
        \item Per filtrare una colonna sulla base dei valori, si combinano slicing e creazione di una nuova colonna
        \begin{itemize}
            \item Lo slicing ha bisogno di una Series di booleani (\texttt{True} se voglio selezionare la riga, \texttt{False} altrimenti)
            \item La nuova colonna ha esattamente ciò
        \end{itemize}

        \bigskip
        \item Vediamo un esempio...
    \end{itemize}
\end{contentframe}

\begin{exampleframe}
    \frametitle{Librerie}
    \framesubtitle{pandas -- Filtrare righe}

    \begin{itemize}
        \item \texttt{selezione = df[df[\textquotesingle{}col1\textquotesingle{}] > 5]}
        \begin{itemize}
            \item \texttt{df[\textquotesingle{}col1\textquotesingle{}] > 5}
            \item Ritorna una Series di booleani
            \item Rappresenta quali righe superano la condizione ($valore > 5$)

            \bigskip
            \item \texttt{selezione = df[...]}
            \item Prende solo le colonne che sono \texttt{True}
        \end{itemize}
    \end{itemize}
\end{exampleframe}
